CREATE TABLE audit_logs (
	id INTEGER NOT NULL, 
	data DATETIME, 
	usuario_id INTEGER, 
	acao VARCHAR(50) NOT NULL, 
	tabela_alvo VARCHAR(50), 
	alvo_id INTEGER, 
	detalhes TEXT, 
	hash_verificacao VARCHAR(64), 
	PRIMARY KEY (id), 
	FOREIGN KEY(usuario_id) REFERENCES usuarios (id)
)
CREATE TABLE categorias (
	id INTEGER NOT NULL, 
	nome VARCHAR(100) NOT NULL, 
	descricao TEXT, 
	PRIMARY KEY (id)
)
CREATE TABLE clientes (
	id INTEGER NOT NULL, 
	codigo VARCHAR(20) NOT NULL, 
	nome VARCHAR(150) NOT NULL, 
	nuit VARCHAR(20), 
	tipo VARCHAR(50), 
	endereco TEXT, 
	cidade VARCHAR(100), 
	provincia VARCHAR(100), 
	telefone VARCHAR(50), 
	telemovel VARCHAR(50), 
	email VARCHAR(150), 
	limite_credito NUMERIC(12, 2), 
	saldo_devedor NUMERIC(12, 2), 
	desconto_padrao NUMERIC(5, 2), 
	pontos_fidelidade NUMERIC(12, 2), 
	ativo BOOLEAN, 
	data_criacao DATETIME, 
	PRIMARY KEY (id), 
	UNIQUE (codigo)
)
CREATE TABLE compras (
